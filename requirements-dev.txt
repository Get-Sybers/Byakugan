#     pip install -e .                        # the package itself, editable

# stage builds and runs go/bin/byakugan-parse.

#   1. bump the pin HERE (and the floor in pyproject.toml only if the new
#      version is now the minimum that works);
#   2. pip install -r requirements-dev.txt
#   3. re-prove the repo:
#        python -m pytest -q
